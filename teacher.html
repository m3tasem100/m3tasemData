`; } // ======================================= // الخطط // ======================================= async function loadPlans(){ content.innerHTML = `

`; const snapshot = await get( ref( db, `users/${user.uid}` ) ); if(!snapshot.exists()) return; const data = snapshot.val(); let html=""; if(!data.subjectIds){ html= ` لا يوجد مواد مرتبطة بك `; } else{ Object.values(data.subjectIds) .forEach(subject=>{ html += `

${subject}

عرض الخطة الدراسية

`; }); } document .getElementById( "plansContainer" ) .innerHTML=html; } // ======================================= // الأسابيع // ======================================= async function loadWeeks(){ content.innerHTML= `

السنة: ${year}

`; }); container.innerHTML=html; } // ======================================= // اعتماد الأسبوع // ======================================= window.verifyWeek= async function(year){ await set( ref( db, `teacherVerification/${user.uid}/${year}` ), { verified:true, date: Date.now() } ); alert( "تم اعتماد الأسبوع" ); } // ======================================= // التقارير // ======================================= function loadReports(){ content.innerHTML= `
